function h(tag, props, ...kids) {
  const el = document.createElement(tag);
  for (const [key, value] of Object.entries(props || {})) {
    if (value == null || value === false) continue;
    if (key === 'class') el.className = value;
    else if (key === 'text') el.textContent = value;
    else if (key === 'dataset') Object.assign(el.dataset, value);
    else if (key.startsWith('on')) el.addEventListener(key.slice(2), value);
    else if (value === true) el.setAttribute(key, '');
    else el.setAttribute(key, value);
  }
  for (const kid of kids.flat(Infinity)) {
    if (kid == null || kid === false) continue;
    el.append(kid.nodeType ? kid : document.createTextNode(String(kid)));
  }
  return el;
}

function parseRoute() {
  const raw = location.hash.replace(/^#\/?/, '');
  const [path, query = ''] = raw.split('?');
  const parts = path.split('/').filter(Boolean).map((p) => { try { return decodeURIComponent(p); } catch (e) { return p; } });
  const page = NAV.some((n) => n.id === parts[0]) || OTHER_PAGES.indexOf(parts[0]) !== -1 ? parts[0] : 'overview';
  return { page, ref: HAS_PANEL.indexOf(page) !== -1 ? parts[1] || '' : '', params: Object.fromEntries(new URLSearchParams(query)) };
}
function hashFor(page, { ref, params } = {}) {
  const query = new URLSearchParams(Object.entries(params || {}).filter(([, v]) => v !== '' && v != null && v !== 'all')).toString();
  return ('#/' + (page) + (ref ? '/' + encodeURIComponent(ref) : '') + (query ? '?' + query : ''));
}
function go(page, options = {}) {

  if (!options.replace) { location.hash = target; return; }

  try { history.replaceState(null, '', target); } catch (err) { location.replace(target); return; }
  route();
}

const drawer = { ref: '', kind: '', node: null, scrim: null, token: 0, draft: null };

function trapFocus(container, event) {
  if (event.key !== 'Tab') return;
  const focusable = [...container.querySelectorAll('a[href], button:not([disabled]), input, textarea, [tabindex]:not([tabindex="-1"])')].filter((el) => el.offsetParent !== null);
  if (!focusable.length) { event.preventDefault(); container.focus(); return; }
  const first = focusable[0], last = focusable[focusable.length - 1];
  if (event.shiftKey && (document.activeElement === first || document.activeElement === container)) { event.preventDefault(); last.focus(); }
  else if (!event.shiftKey && document.activeElement === last) { event.preventDefault(); first.focus(); }
}

function closeDrawerNode() {
  if (!drawer.node) return;
  const ref = drawer.ref;
  drawer.node.remove(); drawer.scrim.remove();
  drawer.node = drawer.scrim = null; drawer.ref = ''; drawer.kind = '';
  document.querySelectorAll('tr.is-open').forEach((tr) => tr.classList.remove('is-open'));
  // The list is redrawn after an action, so the link that opened the panel may be a new element: find its row again.
  const back = state.opener && document.body.contains(state.opener) ? state.opener
    : document.querySelector('tr[data-ref="' + CSS.escape(ref) + '"] .rowlink') || document.querySelector('.rowlink');
  state.opener = null;
  if (back) back.focus({ preventScroll: true });
}
function closeDrawer() {
  if (!drawer.node) return;
  go(state.route.page, { params: state.route.params });
}

function openDrawer(ref) {
  const kind = state.route.page;
  if (drawer.ref === ref && drawer.kind === kind && drawer.node) return;
  const token = ++drawer.token;
  if (!drawer.node) {
    state.opener = document.activeElement;
    drawer.scrim = h('div', { class: 'scrim', onclick: closeDrawer });
    drawer.node = h('div', { class: 'drawer', role: 'dialog', 'aria-modal': 'true', 'aria-label': kind === 'enquiries' ? 'Enquiry details' : 'Enrolment details', tabindex: '-1', onkeydown: (e) => trapFocus(drawer.node, e) });
    $('layer').append(drawer.scrim, drawer.node);
  }
  drawer.ref = ref;
  drawer.kind = kind;
  document.querySelectorAll('tr.is-open').forEach((tr) => tr.classList.remove('is-open'));
  const row = document.querySelector(('tr[data-ref="' + (CSS.escape(ref)) + '"]'));
  if (row) row.classList.add('is-open');
  drawer.node.replaceChildren(
    h('div', { class: 'drawer__head' }, h('div', { class: 'skeleton', style: 'height:24px;width:60%' }), closeButton()),
    h('div', { class: 'drawer__body' }, skeleton(5)));
  drawer.node.focus({ preventScroll: true });
  if (kind === 'enquiries') loadEnquiryDrawer(ref, token); else loadDrawer(ref, token);
}
const closeButton = () => h('button', { class: 'icon-btn', type: 'button', 'aria-label': 'Close details', onclick: closeDrawer }, icon('close'));

function loadDrawer(ref, token) {
  return api('enrolment', { reference: ref }).then((data) => {
    if (token !== drawer.token || !drawer.node) return;
    renderDrawer(data);
  }).catch((err) => {
    if (token !== drawer.token || !drawer.node) return;
    drawer.node.replaceChildren(h('div', { class: 'drawer__head' }, h('h2', { text: 'Could not open this enrolment' }), closeButton()), h('div', { class: 'drawer__body' }, errorBanner(err, /not found/i.test(err.message) ? null : () => loadDrawer(ref, token))));
  });
}

function valueNode(item) {
  if (!item.value) return h('span', { class: 'muted', text: '\u2014' });
  if (item.kind === 'email') return h('a', { href: ('mailto:' + (item.value)), text: item.value });
  if (item.kind === 'phone') {
    const digits = item.value.replace(/\D/g, '');
    return h('span', {}, item.value, digits.length >= 7 ? [' ', h('a', { href: ('https://wa.me/' + (digits)), target: '_blank', rel: 'noopener', text: 'Open in WhatsApp' })] : null);
  }
  if (item.kind === 'mono') return h('span', { class: 'mono', text: item.value });
  return h('span', { text: item.value });
}

/* -- Dialogs ----------------------------------------------- */
function dialog({ title, body, confirm, onConfirm, field }) {
  const opener = document.activeElement;
  const scrim = h('div', { class: 'scrim scrim--dialog' });
  const err = h('p', { class: 'err', role: 'alert' });
  const input = field ? h('input', { id: 'dialog-field', inputmode: 'decimal', autocomplete: 'off', value: field.value }) : null;
  const ok = h('button', { class: 'btn btn--primary', type: 'button', text: confirm });
  const cancel = h('button', { class: 'btn', type: 'button', text: 'Cancel' });
  const node = h('div', { class: 'dialog', role: 'alertdialog', 'aria-modal': 'true', 'aria-labelledby': 'dialog-title', 'aria-describedby': 'dialog-body', tabindex: '-1' },
    h('h2', { id: 'dialog-title', text: title }), h('p', { id: 'dialog-body', text: body }),
    field ? h('label', { for: 'dialog-field', text: field.label }) : null, input, err, h('div', { class: 'dialog__actions' }, cancel, ok));
  const close = () => { scrim.remove(); node.remove(); document.removeEventListener('keydown', onKey, true); if (document.body.contains(opener)) opener.focus({ preventScroll: true }); };
  const onKey = (e) => { if (e.key === 'Escape') { e.stopPropagation(); close(); } else trapFocus(node, e); };
  cancel.addEventListener('click', close);
  scrim.addEventListener('click', close);
  ok.addEventListener('click', async () => {
    err.textContent = ''; ok.disabled = cancel.disabled = true;
    try { await onConfirm(input ? input.value.trim() : ''); close(); }
    catch (e) { err.textContent = e.message; ok.disabled = cancel.disabled = false; }
  });
  document.addEventListener('keydown', onKey, true);
  $('layer').append(scrim, node);
  (input || ok).focus();
}

const clip = (text, n) => { text = String(text || ''); return text.length > n ? text.slice(0, n - 1) + '\u2026' : text; };

const textHas = (item, query, keys) => {
  if (!query) return true;
  const q = query.toLowerCase();

function pagerNode(total, page, size, onPage) {
  const pages = Math.max(1, Math.ceil(total / size));
  const from = total ? (page - 1) * size + 1 : 0;
  const to = Math.min(total, page * size);
  return h('nav', { class: 'pager', 'aria-label': 'Pages' },

// cols: { label, cls, num, mobile (label shown on phones), cell(item) }

// Spreadsheet programs run text that starts with = + - or @ as a formula, so such text is made plain before it goes in a file.
function csvCell(value) {
  let text = value == null ? '' : String(value);
  if (/^[=+\-@\t\r]/.test(text)) text = "'" + text;
  return /[",\n\r]/.test(text) ? '"' + text.replace(/"/g, '""') + '"' : text;
}
function toCsv(headers, rows) {
  return [headers].concat(rows).map((row) => row.map(csvCell).join(',')).join('\r\n') + '\r\n';
}
function downloadCsv(name, text) {
  try {
    const url = URL.createObjectURL(new Blob(['\uFEFF' + text], { type: 'text/csv;charset=utf-8' }));
    const a = h('a', { href: url, download: name });
    document.body.append(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 4000);
    toast('Downloading ' + name + '. If nothing arrives, the page may block downloads: use Copy instead.');
  } catch (err) {
    copy(text, 'The CSV');
  }
}

function showList(page, force) {
  const cfg = PAGES[page];
  setTitle(cfg.title);
  const nonce = state.nonce;
  const cached = state.pages[page];
  if (cached) drawList(page); else mountList(page);
  return api(cfg.method).then((data) => {
    state.pages[page] = data;
    if (page === 'enquiries') { state.openEnquiries = data.counts.open; renderNav(); }
    if (nonce === state.nonce && state.route.page === page) drawList(page);
    // a panel opened from a link needs the list's data; it is drawn once that has arrived
    if (nonce === state.nonce && state.route.ref && drawer.node && drawer.kind === page) loadPanel(state.route.ref);
  }).catch((err) => {
    if (nonce !== state.nonce) return;
    const lp = LISTS[page];
    if (lp && lp.results && document.body.contains(lp.results)) lp.results.replaceChildren(errorBanner(err, () => showList(page, true)));
    else mount(errorBanner(err, () => showList(page, true)));
  });
}

// The toolbar is built once and kept (so typing in the search box is not interrupted); only the results are redrawn.
function mountList(page) {
  const cfg = PAGES[page];
  const lp = LISTS[page] = { mounted: true };
  lp.input = h('input', { id: 'list-search', type: 'search', placeholder: cfg.searchHint, autocomplete: 'off', 'aria-label': cfg.searchLabel });
  let timer;
  lp.input.addEventListener('input', () => {
    clearTimeout(timer);
    timer = setTimeout(() => go(page, { params: Object.assign({}, state.route.params, { q: lp.input.value.trim(), page: 1 }), replace: true }), 220);
  });
  lp.seg = h('div', { class: 'seg', role: 'group', 'aria-label': 'Filter' });
  lp.extra = h('div', { class: 'toolbar__extra' });
  lp.head = h('div');
  lp.results = h('div', { 'aria-live': 'polite' }, skeleton(6));
  mount(
    cfg.intro ? cfg.intro() : null,
    h('div', { class: 'toolbar' }, h('div', { class: 'search' }, icon('search'), lp.input), lp.seg, lp.extra),
    lp.head, lp.results);
}

function drawList(page) {
  const cfg = PAGES[page];
  const data = state.pages[page];
  let lp = LISTS[page];
  if (!lp || !lp.mounted || !document.body.contains(lp.results)) { mountList(page); lp = LISTS[page]; }
  if (!data) return;
  const params = state.route.params;
  const q = (params.q || '').trim();
  const f = params.f || 'all';
  if (document.activeElement !== lp.input) lp.input.value = q;
  const shown = data.items.filter((item) => cfg.matchSeg(item, f) && textHas(item, q, cfg.searchKeys));
  fillSeg(lp.seg, cfg.segments(data), f, (id) => go(page, { params: Object.assign({}, params, { f: id, page: 1 }), replace: true }));
  lp.extra.replaceChildren(...(cfg.extra ? cfg.extra(data, shown) : []));
  lp.head.replaceChildren(...(cfg.head ? [cfg.head(data)] : []));
  if (!shown.length) {
    lp.results.replaceChildren(h('div', { class: 'card' }, h('p', { class: 'empty', text: q || f !== 'all' ? 'Nothing matches.' : cfg.empty })));
    return;
  }
  const pages = Math.max(1, Math.ceil(shown.length / PAGE_SIZE));
  const pageNo = Math.min(Math.max(1, Number(params.page) || 1), pages);
  const slice = shown.slice((pageNo - 1) * PAGE_SIZE, pageNo * PAGE_SIZE);
  lp.results.replaceChildren(
    dataTable(cfg.caption, cfg.columns(), slice, cfg.rowOptions),
    pagerNode(shown.length, pageNo, PAGE_SIZE, (n) => go(page, { params: Object.assign({}, params, { page: n }), replace: true })));
}

function loadPanel(ref) {
  if (state.route.page === 'enquiries') loadEnquiryDrawer(ref, drawer.token); else loadDrawer(ref, drawer.token);

function loadEnquiryDrawer(ref, token) {
  const cached = state.pages.enquiries;
  if (!cached) return; // showList draws the panel once the list has arrived
  if (token !== drawer.token || !drawer.node) return;
  const item = cached.items.find((i) => String(i.row) === String(ref));
  if (!item) {
    drawer.node.replaceChildren(h('div', { class: 'drawer__head' }, h('h2', { text: 'Could not open this enquiry' }), closeButton()), h('div', { class: 'drawer__body' }, errorBanner(new Error('That enquiry was not found. It may have been removed from the sheet.'))));
    return;
  }
  renderEnquiryDrawer(item);
}

      await api('setEnquiryHandled', { row: item.row, email: item.email, at: item.at, handled: !item.handled });

function lookupCard() {
  const input = h('input', { type: 'text', required: true, autocomplete: 'off', spellcheck: 'false', placeholder: 'BAB-XXXX-XXX-1 or UU1-XXXXXXXX' });
  const out = h('div', { 'aria-live': 'polite', style: 'margin-top:14px' });
  const btn = h('button', { class: 'btn btn--primary', type: 'submit', text: 'Check with Squad' });
  const form = h('form', { class: 'lookup', onsubmit: async (e) => {
    e.preventDefault();
    btn.disabled = true;
    out.replaceChildren(h('p', { class: 'muted', text: 'Asking Squad...' }));
    try { renderLookup(out, await api('checkReference', { reference: input.value })); }
    catch (err) { out.replaceChildren(h('p', { class: 'err-line', role: 'alert', text: err.message })); }
    finally { btn.disabled = false; }
  } }, field('Payment reference', input, 'Find it in the Squad dashboard or in a payment email. Looking it up changes nothing.'), btn);
  return h('section', { class: 'card', 'aria-labelledby': 'h-look' }, h('div', { class: 'card__head' }, h('h2', { id: 'h-look', text: 'Look up a payment' })), h('div', { class: 'card__body' }, form, out));
}

function renderLookup(out, r) {
  const stateOf = !r.found ? pill('bad', 'Not found at Squad') : /^success$/i.test(r.status) ? pill('ok', 'Paid') : pill('warn', r.status || 'Not complete');
  const here = r.kind === 'session'
    ? (r.record.alreadyRecorded ? 'Recorded in the Sessions tab' : 'Not recorded yet')
    : (r.record.exists ? r.record.name + ' (' + r.record.reference + ')' + (r.record.alreadyRecorded ? ', payment recorded' : '') : 'No enrolment has this reference');
  const rows = h('ul', { class: 'health' },
    infoRow('Squad says', stateOf),
    infoRow('Mode', pill(r.mode === 'live' ? 'ok' : 'warn', r.mode === 'live' ? 'Live' : 'Sandbox')),
    r.found ? infoRow('Amount', h('span', { text: (r.amount != null ? naira(r.amount) : 'unknown') + ' (expected ' + naira(r.expected) + ')' }), r.amountMatches ? pill('ok', 'Matches') : pill('bad', 'Different')) : null,
    r.found && (r.customer.name || r.customer.email) ? infoRow('Customer', h('span', { text: [r.customer.name, r.customer.email].filter(Boolean).join(', ') })) : null,
    infoRow('In your sheet', h('span', { text: here })),
    r.kind === 'enrolment' && r.record.exists ? infoRow('Open', h('a', { href: hashFor('enrolments', { ref: r.record.reference }), text: 'the enrolment' })) : null);
  const record = r.canRecord ? h('button', { class: 'btn btn--primary', type: 'button', text: 'Record this payment', onclick: () => dialog({
    title: 'Record this payment?',
    body: 'This does what Squad\'s own report would have done: marks the enrolment as paid by Squad and emails you the review link. You still confirm the payment yourself afterwards. Safe to press once; a second press does nothing.',
    confirm: 'Record payment',
    onConfirm: async () => {
      const done = await api('recordPayment', { reference: r.reference });
      const said = done.result === 'ok' ? (done.kind === 'session' ? 'Recorded in the Sessions tab, and you were emailed.' : 'Recorded. The review email has been sent.') : 'Squad answered: ' + done.result + '.';
      out.replaceChildren(h('p', { class: 'callout callout--ok', role: 'status' }, said, done.kind === 'enrolment' ? h('a', { href: hashFor('enrolments', { ref: done.reference }), text: 'Open the enrolment' }) : null));
      refreshData();
    },
  }) }) : null;
  out.replaceChildren(rows, r.reason ? h('p', { class: 'callout callout--warn', role: 'status', text: r.reason }) : null, record ? h('div', { class: 'actions', style: 'margin-top:12px' }, record) : null);
}

/* -- The pages that do not fit in the bottom bar (phones) -- */
function openMore() {
  const opener = document.activeElement;
  const scrim = h('div', { class: 'scrim scrim--dialog' });
  const items = NAV.filter((n) => !n.mobile).concat([{ id: 'new', label: 'Add enrolment', icon: 'plus' }]);
  const node = h('div', { class: 'sheet', role: 'dialog', 'aria-modal': 'true', 'aria-label': 'More pages', tabindex: '-1' },
    h('h2', { text: 'More' }), items.map((n) => h('a', { href: hashFor(n.id), onclick: () => close(true) }, icon(n.icon), n.label)));
  const close = (navigated) => { scrim.remove(); node.remove(); document.removeEventListener('keydown', onKey, true); if (!navigated && document.body.contains(opener)) opener.focus({ preventScroll: true }); };
  const onKey = (e) => { if (e.key === 'Escape') { e.stopPropagation(); close(); } else trapFocus(node, e); };
  scrim.addEventListener('click', () => close());
  document.addEventListener('keydown', onKey, true);
  $('layer').append(scrim, node);
  node.querySelector('a').focus();
}

/* -- Putting it together ----------------------------------- */
function route() {
  const next = parseRoute();
  const prev = state.route;
  state.route = next;
  if (!prev || prev.page !== next.page) {
    state.nonce++;
    enrol.mounted = false;
    Object.keys(LISTS).forEach((k) => { LISTS[k].mounted = false; });
    renderNav();
    closeDrawerNode();
    if (next.page === 'enrolments') showEnrolments();
    else if (next.page === 'new') showNew();
    else if (next.page === 'system') showSystem();
    else if (PAGES[next.page]) showList(next.page);
    else showOverview();
  } else if (next.page === 'enrolments' && !sameParams(prev.params, next.params)) {
    state.nonce++;
    showEnrolments();
  } else if (PAGES[next.page] && !sameParams(prev.params, next.params)) {
    drawList(next.page);
  }
  if (next.ref) openDrawer(next.ref); else closeDrawerNode();
}

function reloadPage(force) {
  const page = state.route.page;
  if (page === 'enrolments') return showEnrolments(force);
  if (page === 'new') return showNew();
  if (page === 'system') return showSystem(force);
  if (PAGES[page]) return showList(page, force);
  return showOverview(force);
}

function boot() {
  applyTheme(stash.get('ba-admin-theme'));
  $('theme-side').addEventListener('click', toggleTheme);
  $('theme-top').addEventListener('click', toggleTheme);
  $('refresh').addEventListener('click', () => {
    state.overview = null; state.list = null; state.pages = {}; state.nonce++;
    const page = state.route.page;
    state.system = null;
    Promise.resolve(reloadPage(true)).then(() => toast('Refreshed'));
  });
  $('search').addEventListener('submit', (e) => { e.preventDefault(); const q = $('search-input').value.trim(); go('enrolments', { params: { q } }); });
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && drawer.node && !document.querySelector('.dialog, .sheet')) closeDrawer();
    if (e.key === '/' && !/^(input|textarea)$/i.test(document.activeElement.tagName) && !e.metaKey && !e.ctrlKey) {
      e.preventDefault();
      const field = $('enrol-search') || $('list-search') || $('search-input');
      field.focus();
    }
  });
  // Coming back to the tab after a while: bring what is on screen up to date, quietly.
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) { state.hiddenAt = Date.now(); return; }
    const away = state.hiddenAt ? Date.now() - state.hiddenAt : 0;
    state.hiddenAt = 0;
    if (away > 5 * 60 * 1000 && state.route && state.route.page !== 'new' && !document.querySelector('.dialog')) reloadPage(true);
  });
  window.addEventListener('hashchange', route);
  $('user').textContent = '';
  route();
  // the signed-in address arrives with the first answer
  const waitForUser = setInterval(() => { if (state.user) { $('user').textContent = state.user; clearInterval(waitForUser); } }, 300);
  setTimeout(() => clearInterval(waitForUser), 15000);
}
try { boot(); } catch (err) { showFatal(err); }
})();
