<script>
// Runs before the dashboard script. If that script cannot be parsed, throws before it draws anything, or is blocked,
// this says so on the page instead of leaving it empty.
(function () {
  function say(text) {
    var box = document.getElementById('boot');
    if (!box) return;
    box.className = 'boot boot--bad';
    box.textContent = text;
  }
  window.addEventListener('error', function (e) {
    say('The dashboard could not start: ' + (e.message || 'script error') + (e.lineno ? ' (line ' + e.lineno + ', column ' + (e.colno || '?') + ')' : ''));
    // Show the text of that line exactly as this page received it; a copy that was altered on the way shows up here.
    try {
      var box = document.getElementById('boot'), main = '';
      for (var i = 0; i < document.scripts.length; i++) if (document.scripts[i].text.length > main.length) main = document.scripts[i].text;
      var line = main.split('\n')[(e.lineno || 0) - 1], col = e.colno || 1;
      if (box && line) {
        var src = document.createElement('pre');
        src.className = 'boot__src';
        src.textContent = 'Received, line ' + e.lineno + ': ' + line.slice(Math.max(0, col - 70), col + 90);
        box.appendChild(src);
      }
    } catch (ignore) { /* the message above is still shown */ }
  });
  document.addEventListener('securitypolicyviolation', function (e) { say('The page was blocked by a security rule: ' + e.violatedDirective + ' (' + (e.blockedURI || 'inline') + ')'); });
})();
</script>

function h(tag, props, ...kids) {
  const el = document.createElement(tag);
  for (const [key, value] of Object.entries(props || {})) {
    if (value == null || value === false) continue;
    if (key === 'class') el.className = value;
    else if (key === 'text') el.textContent = value;
    else if (key.startsWith('on')) el.addEventListener(key.slice(2), value);
    else if (value === true) el.setAttribute(key, '');
    else el.setAttribute(key, value);
  }
  for (const kid of kids.flat(Infinity)) {
    if (kid == null || kid === false) continue;
    el.append(kid.nodeType ? kid : document.createTextNode(String(kid)));
  }
  return el;
}

function parseRoute() {
  const raw = location.hash.replace(/^#\/?/, '');
  const [path, query = ''] = raw.split('?');
  const parts = path.split('/').filter(Boolean).map((p) => { try { return decodeURIComponent(p); } catch (e) { return p; } });
  const page = NAV.some((n) => n.id === parts[0]) ? parts[0] : 'overview';
  return { page, ref: page === 'enrolments' ? parts[1] || '' : '', params: Object.fromEntries(new URLSearchParams(query)) };
}
function hashFor(page, { ref, params } = {}) {
  const query = new URLSearchParams(Object.entries(params || {}).filter(([, v]) => v !== '' && v != null && v !== 'all')).toString();
  return ('#/' + (page) + (ref ? '/' + encodeURIComponent(ref) : '') + (query ? '?' + query : ''));
}
function go(page, options = {}) {
  const target = hashFor(page, options);
  if (!options.replace) { location.hash = target; return; }
  // Changing filters or pages replaces the address instead of adding to Back history. If a frame refuses replaceState, location.replace does the same and fires hashchange.
  try { history.replaceState(null, '', target); } catch (err) { location.replace(target); return; }
  route();
}

function trapFocus(container, event) {
  if (event.key !== 'Tab') return;
  const focusable = [...container.querySelectorAll('a[href], button:not([disabled]), input, textarea, [tabindex]:not([tabindex="-1"])')].filter((el) => el.offsetParent !== null);
  if (!focusable.length) { event.preventDefault(); container.focus(); return; }
  const first = focusable[0], last = focusable[focusable.length - 1];
  if (event.shiftKey && (document.activeElement === first || document.activeElement === container)) { event.preventDefault(); last.focus(); }
  else if (!event.shiftKey && document.activeElement === last) { event.preventDefault(); first.focus(); }
}

function closeDrawerNode() {
  if (!drawer.node) return;
  drawer.node.remove(); drawer.scrim.remove();
  drawer.node = drawer.scrim = null; drawer.ref = '';
  document.querySelectorAll('tr.is-open').forEach((tr) => tr.classList.remove('is-open'));
  const back = state.opener && document.body.contains(state.opener) ? state.opener : document.querySelector('.rowlink');
  state.opener = null;
  if (back) back.focus({ preventScroll: true });
}
function closeDrawer() {
  if (!drawer.node) return;
  go('enrolments', { params: state.route.params });
}

function openDrawer(ref) {
  if (drawer.ref === ref && drawer.node) return;
  const token = ++drawer.token;
  if (!drawer.node) {
    state.opener = document.activeElement;
    drawer.scrim = h('div', { class: 'scrim', onclick: closeDrawer });
    drawer.node = h('div', { class: 'drawer', role: 'dialog', 'aria-modal': 'true', 'aria-label': 'Enrolment details', tabindex: '-1', onkeydown: (e) => trapFocus(drawer.node, e) });
    $('layer').append(drawer.scrim, drawer.node);
  }
  drawer.ref = ref;
  document.querySelectorAll('tr.is-open').forEach((tr) => tr.classList.remove('is-open'));
  const row = document.querySelector(('tr[data-ref="' + (CSS.escape(ref)) + '"]'));
  if (row) row.classList.add('is-open');
  drawer.node.replaceChildren(
    h('div', { class: 'drawer__head' }, h('div', { class: 'skeleton', style: 'height:24px;width:60%' }), closeButton()),
    h('div', { class: 'drawer__body' }, skeleton(5)));
  drawer.node.focus({ preventScroll: true });
  loadDrawer(ref, token);
}
const closeButton = () => h('button', { class: 'icon-btn', type: 'button', 'aria-label': 'Close details', onclick: closeDrawer }, icon('close'));

function loadDrawer(ref, token) {
  return api('enrolment', { reference: ref }).then((data) => {
    if (token !== drawer.token || !drawer.node) return;
    renderDrawer(data);
  }).catch((err) => {
    if (token !== drawer.token || !drawer.node) return;
    drawer.node.replaceChildren(h('div', { class: 'drawer__head' }, h('h2', { text: 'Could not open this enrolment' }), closeButton()), h('div', { class: 'drawer__body' }, errorBanner(err, /not found/i.test(err.message) ? null : () => loadDrawer(ref, token))));
  });
}

function renderDrawer(data) {
  const s = data.summary;
  const act = (call, success) => async (event) => {
    const btn = event.currentTarget;
    btn.disabled = true;
    try { success(await call()); refreshData(); loadDrawer(s.reference, drawer.token); }
    catch (err) { toast(err.message, true); }
    finally { btn.disabled = false; }
  };
  const actions = h('div', { class: 'actions' },
    h('button', { class: 'btn', type: 'button', onclick: act(() => api('sendReview', { reference: s.reference }), (r) => toast(r.sent ? ('Review email sent to ' + (r.to)) : ('Review email not sent: ' + (r.error)), !r.sent)), text: 'Send review email' }),
    data.canConfirm ? h('button', { class: 'btn btn--primary', type: 'button', onclick: () => confirmDialog(data), text: 'Confirm payment' }) : null,
    data.canResend ? h('button', { class: 'btn', type: 'button', onclick: () => resendDialog(data), text: 'Resend welcome email' }) : null);

  const sections = data.sections.map((sec) => h('section', { class: 'sec' }, h('h3', { text: sec.title }), h('dl', { class: 'kv' },
    sec.items.map((item) => [h('dt', { text: item.label }), h('dd', {}, valueNode(item), item.value && (item.kind === 'email' || item.kind === 'mono') ? copyButton(item.value, item.label.toLowerCase()) : null)]))));

  const note = h('textarea', { id: 'note', maxlength: String(data.notesMax), 'aria-label': 'Admin notes', placeholder: 'Only you can see these notes.' });
  // The drawer is redrawn after every action; whatever is typed in the note and not yet saved must survive that.
  note.value = drawer.draft && drawer.draft.ref === s.reference ? drawer.draft.value : data.notes || '';
  const counter = h('span', { text: ('' + (note.value.length) + '/' + (data.notesMax)) });
  note.addEventListener('input', () => { counter.textContent = ('' + (note.value.length) + '/' + (data.notesMax)); drawer.draft = { ref: s.reference, value: note.value }; });
  const save = h('button', { class: 'btn btn--sm', type: 'button', text: 'Save note', onclick: async (e) => {
    const btn = e.currentTarget; // currentTarget is cleared once the handler has awaited
    btn.disabled = true;
    try { await api('saveNote', { reference: s.reference, note: note.value }); drawer.draft = null; toast('Note saved'); }
    catch (err) { toast(err.message, true); }
    finally { btn.disabled = false; }
  } });

  drawer.node.setAttribute('aria-label', ('Enrolment: ' + (s.name || s.reference)));
  drawer.node.replaceChildren(
    h('div', { class: 'drawer__head' }, h('div', {}, h('h2', { text: s.name || s.reference }), h('p', { class: 'muted mono', text: s.reference })), closeButton()),
    h('div', { class: 'drawer__body' },
      h('div', {}, statusBadge(s)),
      actions,
      sections,
      h('section', { class: 'sec note' }, h('h3', { text: 'Admin notes' }), note, h('div', { class: 'note__row' }, counter, save))));
}

function valueNode(item) {
  if (!item.value) return h('span', { class: 'muted', text: '\u2014' });
  if (item.kind === 'email') return h('a', { href: ('mailto:' + (item.value)), text: item.value });
  if (item.kind === 'phone') {
    const digits = item.value.replace(/\D/g, '');
    return h('span', {}, item.value, digits.length >= 7 ? [' ', h('a', { href: ('https://wa.me/' + (digits)), target: '_blank', rel: 'noopener', text: 'Open in WhatsApp' })] : null);
  }
  if (item.kind === 'mono') return h('span', { class: 'mono', text: item.value });
  return h('span', { text: item.value });
}

/* -- Dialogs ----------------------------------------------- */
function dialog({ title, body, confirm, onConfirm, field }) {
  const opener = document.activeElement;
  const scrim = h('div', { class: 'scrim scrim--dialog' });
  const err = h('p', { class: 'err', role: 'alert' });
  const input = field ? h('input', { id: 'dialog-field', inputmode: 'decimal', autocomplete: 'off', value: field.value }) : null;
  const ok = h('button', { class: 'btn btn--primary', type: 'button', text: confirm });
  const cancel = h('button', { class: 'btn', type: 'button', text: 'Cancel' });
  const node = h('div', { class: 'dialog', role: 'alertdialog', 'aria-modal': 'true', 'aria-labelledby': 'dialog-title', 'aria-describedby': 'dialog-body', tabindex: '-1' },
    h('h2', { id: 'dialog-title', text: title }), h('p', { id: 'dialog-body', text: body }),
    field ? h('label', { for: 'dialog-field', text: field.label }) : null, input, err, h('div', { class: 'dialog__actions' }, cancel, ok));
  const close = () => { scrim.remove(); node.remove(); document.removeEventListener('keydown', onKey, true); if (document.body.contains(opener)) opener.focus({ preventScroll: true }); };
  const onKey = (e) => { if (e.key === 'Escape') { e.stopPropagation(); close(); } else trapFocus(node, e); };
  cancel.addEventListener('click', close);
  scrim.addEventListener('click', close);
  ok.addEventListener('click', async () => {
    err.textContent = ''; ok.disabled = cancel.disabled = true;
    try { await onConfirm(input ? input.value.trim() : ''); close(); }
    catch (e) { err.textContent = e.message; ok.disabled = cancel.disabled = false; }
  });
  document.addEventListener('keydown', onKey, true);
  $('layer').append(scrim, node);
  (input || ok).focus();
}

function confirmDialog(data) {
  const s = data.summary;
  dialog({
    title: 'Confirm payment',
    body: ('This marks ' + (s.name || s.reference) + ' as paid and emails them the welcome message with the WhatsApp group link, copied to ' + (data.welcomeCc.split(',').join(' and ')) + '. Compare the payment with the Squad dashboard first.'),
    field: { label: 'Amount paid (NGN)', value: data.defaultAmount },
    confirm: 'Confirm and send welcome email',
    onConfirm: async (amount) => {
      const r = await api('confirmPayment', { reference: s.reference, amountPaid: amount });
      toast(r.welcome === 'failed' ? ('Confirmed, but the welcome email failed: ' + (r.error)) : ('Confirmed. Welcome email sent to ' + (r.to)), r.welcome === 'failed');
      refreshData(); loadDrawer(s.reference, drawer.token);
    },
  });
}
function resendDialog(data) {
  const s = data.summary;
  dialog({
    title: 'Send the welcome email again?',
    body: ('' + (s.name || s.reference) + ' already has one. This sends it again to ' + (s.email) + ', copied to ' + (data.welcomeCc.split(',').join(' and ')) + '.'),
    confirm: 'Send again',
    onConfirm: async () => {
      const r = await api('resendWelcome', { reference: s.reference });
      toast(r.welcome === 'sent' ? ('Welcome email sent to ' + (r.to)) : ('Not sent: ' + (r.error)), r.welcome !== 'sent');
      refreshData(); loadDrawer(s.reference, drawer.token);
    },
  });
}

/* -- Putting it together ----------------------------------- */
function route() {
  const next = parseRoute();
  const prev = state.route;
  state.route = next;
  if (!prev || prev.page !== next.page) {
    state.nonce++;
    enrol.mounted = false;
    renderNav();
    closeDrawerNode();
    (next.page === 'enrolments' ? showEnrolments() : showOverview());
  } else if (next.page === 'enrolments' && !sameParams(prev.params, next.params)) {
    state.nonce++;
    showEnrolments();
  }
  if (next.ref) openDrawer(next.ref); else closeDrawerNode();
}

function boot() {
  applyTheme(stash.get('ba-admin-theme'));
  $('theme-side').addEventListener('click', toggleTheme);
  $('theme-top').addEventListener('click', toggleTheme);
  $('refresh').addEventListener('click', () => { state.overview = null; state.list = null; state.nonce++; (state.route.page === 'enrolments' ? showEnrolments(true) : showOverview(true)).then(() => toast('Refreshed')); });
  $('search').addEventListener('submit', (e) => { e.preventDefault(); const q = $('search-input').value.trim(); go('enrolments', { params: { q } }); });
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && drawer.node && !document.querySelector('.dialog')) closeDrawer();
    if (e.key === '/' && !/^(input|textarea)$/i.test(document.activeElement.tagName) && !e.metaKey && !e.ctrlKey) {
      e.preventDefault();
      const field = enrol.input && $('enrol-search') ? enrol.input : $('search-input');
      field.focus();
    }
  });
  window.addEventListener('hashchange', route);
  $('user').textContent = '';
  route();
  // the signed-in address arrives with the first answer
  const waitForUser = setInterval(() => { if (state.user) { $('user').textContent = state.user; clearInterval(waitForUser); } }, 300);
  setTimeout(() => clearInterval(waitForUser), 15000);
}
try { boot(); } catch (err) { showFatal(err); }
})();
