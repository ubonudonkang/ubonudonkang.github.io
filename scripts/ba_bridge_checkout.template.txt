DORMANT REOPENING TEMPLATE — BA Bridge enrolment flow

This is reference source, stored as plain text so it cannot run as a website page.
It contains the enrolment form and its Apps Script + Selar checkout handler.
Use open_ba_registration.mjs with complete cohort details to render it.

Check the deployed Apps Script, Selar product price and post-purchase return URL
before publishing. A browser return is not payment proof.

<script>
  /* =====================================================================
     This deployed Apps Script Web app URL must accept action=signup.
     ===================================================================== */
  var ENDPOINT_URL = "{{APPS_SCRIPT_URL}}";

  (function(){
    var form = document.getElementById("waitlist-form");
    var statusEl = document.getElementById("form-status");
    var submitBtn = document.getElementById("submit-btn");
    var doneEl = document.getElementById("form-done");

    /* "Other" text inputs: reveal when Other is picked */
    function toggleOther(control, targetId, active){
      var box = document.getElementById(targetId);
      if(!box) return;
      box.hidden = !active;
      if(active){ box.focus(); } else { box.value = ""; }
    }
    document.querySelectorAll("select[data-other]").forEach(function(sel){
      sel.addEventListener("change", function(){
        toggleOther(sel, sel.dataset.other, sel.value === "Other");
      });
    });
    document.querySelectorAll("input[type=checkbox][data-other]").forEach(function(box){
      box.addEventListener("change", function(){
        toggleOther(box, box.dataset.other, box.checked);
      });
    });

    /* Checkbox pills: visual state, and cap on multi-select groups */
    document.querySelectorAll(".f-checks").forEach(function(group){
      var max = parseInt(group.dataset.max || "0", 10);
      group.addEventListener("change", function(){
        var boxes = group.querySelectorAll("input[type=checkbox]");
        var checked = group.querySelectorAll("input[type=checkbox]:checked").length;
        boxes.forEach(function(box){
          var pill = box.closest(".f-check");
          pill.classList.toggle("checked", box.checked);
          if(max){
            var atMax = checked >= max && !box.checked;
            box.disabled = atMax;
            pill.classList.toggle("maxed", atMax);
          }
        });
      });
    });
    document.querySelectorAll(".f-consent input").forEach(function(box){
      box.addEventListener("change", function(){
        box.closest(".f-check").classList.toggle("checked", box.checked);
      });
    });

    function showError(msg, el){
      statusEl.textContent = msg;
      statusEl.classList.remove("ok");
      if(el){
        el.classList.add("invalid");
        el.focus({ preventScroll:false });
        el.addEventListener("input", function once(){ el.classList.remove("invalid"); el.removeEventListener("input", once); }, false);
      }
    }

    form.addEventListener("submit", function(e){
      e.preventDefault();
      statusEl.textContent = "";

      /* Validation */
      var required = form.querySelectorAll("[required]");
      for(var i = 0; i < required.length; i++){
        var el = required[i];
        if(el.type === "checkbox"){
          if(!el.checked){ showError("Please tick the consent box to continue.", el); return; }
        } else if(!el.value.trim()){
          showError("Please fill in every field marked with a star.", el); return;
        } else if(el.type === "email" && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(el.value)){
          showError("That email address does not look right.", el); return;
        }
      }
      var wants = form.querySelectorAll("input[name=wants]:checked");
      if(wants.length === 0){ showError("Pick at least one thing you want from the program."); return; }
      var avail = form.querySelectorAll("input[name=availability]:checked");
      if(avail.length === 0){ showError("Pick at least one time you could attend."); return; }

      if(ENDPOINT_URL.indexOf("http") !== 0){
        showError("This form is not connected yet. Set ENDPOINT_URL in the page source.");
        return;
      }

      /* Collect: checkbox groups join into one comma-separated value */
      var payload = new URLSearchParams();
      var raw = new FormData(form);
      var grouped = {};
      raw.forEach(function(value, key){
        if(grouped[key] === undefined){ grouped[key] = value; }
        else { grouped[key] = grouped[key] + ", " + value; }
      });

      /* Fold any "Other" text into its parent answer, so the sheet keeps
         one column per question. */
      ["background", "wants", "budget", "source"].forEach(function(key){
        var extra = (grouped[key + "_other"] || "").trim();
        delete grouped[key + "_other"];
        if(extra && grouped[key]){
          grouped[key] = grouped[key].replace(/\bOther\b/, "Other: " + extra);
        }
      });

      /* The application is saved before checkout so it can be reconciled with a Selar sale. */
      var ref = "BAB-" + Date.now().toString(36).toUpperCase() + "-" + Math.random().toString(36).toUpperCase().replace(/[^A-Z0-9]/g, "").slice(0, 3);
      var buyer = { name: (grouped.name || "").trim(), email: (grouped.email || "").trim(), whatsapp: grouped.whatsapp || "" };
      grouped.reference = ref;
      grouped.amount = "{{PRICE_NGN}}";
      grouped.cohort = "{{COHORT}}";
      grouped.payment_status = "Awaiting payment";
      grouped.action = "signup";

      Object.keys(grouped).forEach(function(k){ payload.append(k, grouped[k]); });
      submitBtn.disabled = true;
      submitBtn.textContent = "Saving...";

      post(payload)
        .then(function(){
          rememberPending({ reference: ref, name: buyer.name, email: buyer.email, whatsapp: buyer.whatsapp, product: "ba-bridge-selar", cohort: "{{COHORT}}" });
          submitBtn.hidden = true;
          document.getElementById("done-ref").textContent = ref;
          if (window.uuTrack) window.uuTrack("cohort-enrolment-started");
          startPayment();
        })
        .catch(function(){
          submitBtn.disabled = false;
          submitBtn.textContent = "Continue to payment";
          showError("That did not go through. Please try again. Cohort enquiries: ubonanalyst@gmail.com.");
        });
    });

    /* ---------- Payment: hosted Selar checkout ---------- */
    var retryBtn = document.getElementById("pay-retry");
    var SELAR_COHORT_URL = "{{SELAR_URL}}";
    var PENDING_KEY = "uu-cohort-selar-{{COHORT_SLUG}}-pending";
    var RETURNED_KEY = "uu-cohort-selar-{{COHORT_SLUG}}-returned";
    var LEGACY_PENDING_KEY = "uu-cohort-selar-pending";

    function validPending(value){
      return value && value.product === "ba-bridge-selar" && value.cohort === "{{COHORT}}" && typeof value.reference === "string" &&
        /^BAB-/.test(value.reference) && typeof value.name === "string" && value.name &&
        typeof value.email === "string" && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.email);
    }
    function readPending(){
      try { var value = JSON.parse(sessionStorage.getItem(PENDING_KEY) || "null"); return validPending(value) ? value : null; }
      catch(e) { return null; }
    }
    function readReturned(){
      try { var value = JSON.parse(sessionStorage.getItem(RETURNED_KEY) || "null"); return validPending(value) ? value : null; }
      catch(e) { return null; }
    }
    function readLegacy(key){
      try {
        var value = JSON.parse(sessionStorage.getItem(key) || "null");
        return value && value.product === "ba-bridge-selar" && !value.cohort && /^BAB-/.test(value.reference || "") &&
          /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.email || "") ? value : null;
      } catch(e) { return null; }
    }
    function rememberPending(value){
      try { sessionStorage.setItem(PENDING_KEY, JSON.stringify(value)); } catch(e) { /* checkout remains available */ }
    }
    function removePaymentParam(){
      var url = new URL(window.location.href);
      if(!url.searchParams.has("payment")) return;
      url.searchParams.delete("payment");
      var query = url.searchParams.toString();
      window.history.replaceState({}, "", url.pathname + (query ? "?" + query : "") + url.hash);
    }
    function selarUrl(customer){
      var url = new URL(SELAR_COHORT_URL);
      url.searchParams.set("add_to_cart", "1");
      url.searchParams.set("email", customer.email);
      url.searchParams.set("fullname", customer.name);
      if(customer.whatsapp) url.searchParams.set("mobile", customer.whatsapp);
      return url.toString();
    }
    function showReturnedPayment(customer){
      document.getElementById("done-ref").textContent = customer.reference;
      statusEl.textContent = "";
      form.hidden = true;
      retryBtn.hidden = true;
      doneEl.hidden = false;
      doneEl.scrollIntoView({ behavior:"smooth", block:"center" });
      doneEl.focus({ preventScroll:true });
    }
    function startPayment(){
      var pending = readPending();
      if(!pending){
        statusEl.classList.remove("ok");
        statusEl.textContent = "We could not restore your saved enrolment. Please submit the form again.";
        return;
      }
      retryBtn.hidden = true;
      statusEl.classList.add("ok");
      statusEl.textContent = "Taking you to secure payment...";
      if (window.uuTrack) window.uuTrack("cohort-selar-checkout-opened");
      window.location.assign(selarUrl(pending));
    }
    retryBtn.addEventListener("click", startPayment);

    (function restoreFlow(){
      var returned = new URLSearchParams(window.location.search).get("payment") === "selar";
      var pending = readPending();
      if(returned){
        removePaymentParam();
        if(!pending){
          var earlier = readLegacy(LEGACY_PENDING_KEY);
          if(earlier){
            showReturnedPayment(earlier);
            return;
          }
          statusEl.textContent = "We could not restore this payment return. Please contact us if you completed payment.";
          return;
        }
        try { sessionStorage.setItem(RETURNED_KEY, JSON.stringify(pending)); } catch(e) {}
        showReturnedPayment(pending);
        if (window.uuTrack) window.uuTrack("cohort-selar-returned");
        return;
      }
      var returnedState = readReturned();
      if(returnedState){
        showReturnedPayment(returnedState);
        return;
      }
      if(pending){
        submitBtn.hidden = true;
        document.getElementById("done-ref").textContent = pending.reference;
        statusEl.classList.add("ok");
        statusEl.textContent = "Your enrolment is saved. Continue to secure payment when you are ready.";
        retryBtn.hidden = false;
      }
    })();
    /* Posts to the endpoint, falling back to a blind send if the browser
       will not let us read the response. */
    function post(body){
      return fetch(ENDPOINT_URL, { method:"POST", body:body })
        .then(function(r){ return r.json(); })
        .then(function(res){
          if(!res || res.result !== "ok"){ throw new Error("rejected"); }
        })
        .catch(function(err){
          if(err && err.message === "rejected"){ throw err; }
          return fetch(ENDPOINT_URL, { method:"POST", mode:"no-cors", body:body }).then(function(){ return; });
        });
    }

    var items = document.querySelectorAll(".reveal");
    if(!("IntersectionObserver" in window)){
      items.forEach(function(el){ el.classList.add("in"); });
      return;
    }
    var io = new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        if(entry.isIntersecting){
          entry.target.classList.add("in");
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.12, rootMargin: "0px 0px -40px 0px" });
    items.forEach(function(el){ io.observe(el); });
  })();
</script>
