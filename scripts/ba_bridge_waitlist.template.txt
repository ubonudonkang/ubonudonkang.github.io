It disables new checkout on the page while preserving earlier payment returns.

<script>
  /* Keep the return message available to people who completed an earlier
     checkout. A browser return is not verified payment. */
  (function(){
    var form = document.getElementById('waitlist-form');
    var done = document.getElementById('form-done');
    var status = document.getElementById('payment-status');
    var pendingKey = 'uu-cohort-selar-pending';
    var returnedKey = 'uu-cohort-selar-returned';

    function valid(value){
      return value && value.product === 'ba-bridge-selar' &&
        typeof value.reference === 'string' && /^BAB-/.test(value.reference) &&
        typeof value.name === 'string' && value.name &&
        typeof value.email === 'string' && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.email);
    }
    function read(key){
      try { var value = JSON.parse(sessionStorage.getItem(key) || 'null'); return valid(value) ? value : null; }
      catch(e) { return null; }
    }
    function removePaymentParam(){
      var url = new URL(window.location.href);
      if(!url.searchParams.has('payment')) return;
      url.searchParams.delete('payment');
      var query = url.searchParams.toString();
      window.history.replaceState({}, '', url.pathname + (query ? '?' + query : '') + url.hash);
    }
    function showEarlierReturn(customer){
      document.getElementById('done-ref').textContent = customer.reference;
      form.hidden = true;
      done.hidden = false;
      done.focus({ preventScroll: true });
    }

    var isReturn = new URLSearchParams(window.location.search).get('payment') === 'selar';
    var pending = read(pendingKey);
    if(isReturn){
      removePaymentParam();
      if(pending){
        try { sessionStorage.setItem(returnedKey, JSON.stringify(pending)); } catch(e) {}
        showEarlierReturn(pending);
        if(window.uuTrack) window.uuTrack('cohort-selar-returned');
      } else {
        status.textContent = 'If you completed an earlier payment, please contact us with your Selar receipt so we can check it.';
      }
    } else {
      var earlierReturn = read(returnedKey);
      if(earlierReturn) showEarlierReturn(earlierReturn);
      else if(pending) status.textContent = 'Registration is closed. If you already paid, contact us with reference ' + pending.reference + '.';
    }

    var items = document.querySelectorAll('.reveal');
    if(!('IntersectionObserver' in window)){
      items.forEach(function(el){ el.classList.add('in'); });
      return;
    }
    var io = new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        if(entry.isIntersecting){
          entry.target.classList.add('in');
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
    items.forEach(function(el){ io.observe(el); });
  })();
  window.formspree = window.formspree || function () { (formspree.q = formspree.q || []).push(arguments); };
  formspree('initForm', { formElement: '#waitlist-form', formId: 'xdaqwayj' });
</script>
<script src="https://unpkg.com/@formspree/ajax@1" defer></script>
